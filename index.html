function executarBuscaNoSistema() { const placaInput = document.getElementById("placa").value.trim().toUpperCase(); const modeloInput = document.getElementById("modelo-input").value.trim().toUpperCase(); const categoriaSelect = document.getElementById("categoria").value; const loadingDiv = document.getElementById("loading"); const resultadoDiv = document.getElementById("resultado"); if (!categoriaSelect) { alert("Por favor, selecione uma categoria de peças."); return; } loadingDiv.style.display = "block"; resultadoDiv.style.display = "none"; resultadoDiv.innerHTML = ""; setTimeout(() => { loadingDiv.style.display = "none"; resultadoDiv.style.display = "block"; // Define qual chave do objeto usar (procura por UNO, CIVIC ou usa PADRAO) let veiculoChave = "PADRAO"; if (placaInput === "ALK1183" || modeloInput.includes("UNO")) { veiculoChave = "UNO"; } else if (modeloInput.includes("CIVIC")) { veiculoChave = "CIVIC"; } const dadosVeiculo = catalogoPecasTecnicas[veiculoChave] || catalogoPecasTecnicas["PADRAO"]; let htmlConteudo = `
Marca: ${dadosVeiculo.marca}
Modelo: ${dadosVeiculo.modelo}
Ano/Versao: ${dadosVeiculo.ano || dadosVeiculo.versao}
Categoria Selecionada: ${categoriaSelect.toUpperCase()}
`; if (categoriaSelect === "oleo_filtros" && dadosVeiculo.oleo_filtros) { const oleo = dadosVeiculo.oleo_filtros; htmlConteudo += `

Viscosidade: ${oleo.viscosidade}

Quantidade: ${oleo.quantidade}

`; oleo.filtros.forEach(f => { htmlConteudo += ``; }); htmlConteudo += `
FiltroMarcaCódigo
${f.nome}${f.marca}${f.codigo}
`; } else if (dadosVeiculo[categoriaSelect] && dadosVeiculo[categoriaSelect].pecas) { htmlConteudo += ` `; dadosVeiculo[categoriaSelect].pecas.forEach(p => { htmlConteudo += ``; }); htmlConteudo += `
Peça / ItemMarcaCódigo
${p.item}${p.marca}${p.codigo}
`; } else { htmlConteudo += `

Nenhum detalhe encontrado para esta categoria.

`; } resultadoDiv.innerHTML = htmlConteudo; }, 800); }
